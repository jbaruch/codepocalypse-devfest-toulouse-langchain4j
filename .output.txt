### Retrieval Augmented Generation (RAG) Example

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/integration-tests/simple-ollama/src/main/resources/META-INF/resources/index.html

Example of implementing Retrieval Augmented Generation (RAG) using Langchain4j and Quarkus. This involves a retriever and a response generator.

```Java
import jakarta.inject.Inject;
import jakarta.ws.rs.GET;
import jakarta.ws.rs.Path;
import jakarta.ws.rs.QueryParam;
import dev.langchain4j.data.document.Document;
import dev.langchain4j.data.embedding.Embedding;
import dev.langchain4j.data.segment.TextSegment;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.embedding.EmbeddingModel;
import dev.langchain4j.retriever.Retriever;
import dev.langchain4j.service.AiServices;
import dev.langchain4j.store.embedding.EmbeddingStore;

import java.util.List;

@Path("/rag")
public class RagResource {

    @Inject
    ChatLanguageModel chatLanguageModel;

    @Inject
    EmbeddingModel embeddingModel;

    @Inject
    EmbeddingStore<TextSegment> embeddingStore;

    @Inject
    Retriever<TextSegment> retriever;

    @GET
    @Path("/query")
    public String query(@QueryParam("question") String question) {
        // Add documents to the store (typically done during application startup or data ingestion)
        // addDocument("Your document content here...");

        // Create a RAG chain
        RagChain ragChain = AiServices.create(RagChain.class,
                chatLanguageModel,
                retriever);

        return ragChain.answer(question);
    }

    // Helper method to add documents (example)
    private void addDocument(String text) {
        Document document = Document.from(text);
        List<Embedding> embeddings = embeddingModel.embed(document.textSegments());
        embeddingStore.addAll(embeddings, document.textSegments());
    }

    // Define the RAG interface
    interface RagChain {
        String answer(String question);
    }
}
```

### Retrieval Augmented Generation (RAG) Example

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/integration-tests/simple-ollama/src/main/resources/META-INF/resources/index.html

This example illustrates how to implement Retrieval Augmented Generation (RAG) using Langchain4j in Quarkus. It involves setting up a retriever, a prompt template, and a chain to generate responses based on retrieved documents.

```Java
import jakarta.inject.Inject;
import dev.langchain4j.data.document.DocumentLoader;
import dev.langchain4j.data.document.FileSystemDocumentLoader;
import dev.langchain4j.data.embedding.DocumentTransformer;
import dev.langchain4j.data.segment.TextSegment;
import dev.langchain4j.model.embedding.EmbeddingModel;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.retriever.Retriever;
import dev.langchain4j.service.AiServices;
import dev.langchain4j.data.document.splitter.DocumentSplitter;
import dev.langchain4j.data.document.splitter.ParagraphSplitter;
import dev.langchain4j.store.embedding.EmbeddingStore;
import dev.langchain4j.store.embedding.InMemoryEmbeddingStore;
import dev.langchain4j.model.output.OutputParser;
import dev.langchain4j.model.output.OutputParsers;

import java.nio.file.Paths;

public class RagService {

    @Inject
    ChatLanguageModel chatModel;

    @Inject
    EmbeddingModel embeddingModel;

    private Retriever<TextSegment> retriever;

    public void setupRAG() {
        // Load documents from file system
        DocumentLoader documentLoader = new FileSystemDocumentLoader(Paths.get("src/main/resources/documents"));
        List<Document> documents = documentLoader.load();

        // Split documents into segments
        DocumentSplitter splitter = new ParagraphSplitter();
        List<TextSegment> textSegments = splitter.split(documents);

        // Create an embedding store and add segments
        EmbeddingStore<TextSegment> embeddingStore = new InMemoryEmbeddingStore<>();
        embeddingStore.addAll(textSegments, embeddingModel.embedAll(textSegments));

        // Create a retriever
        this.retriever = new EmbeddingStoreRetriever<>(embeddingStore, embeddingModel);
    }

    public String answerQuestion(String question) {
        // Define a RAG chain
        RAGChain<TextSegment> ragChain = RAGChain.builder()
                .chatLanguageModel(chatModel)
                .retriever(retriever)
                .promptTemplate("Answer the following question based on the provided context:\n\n{context}\n\nQuestion: {question}")
                .outputParser(OutputParsers.STRING_OUTPUT_PARSER)
                .build();

        return ragChain.execute(question);
    }
}
```

### RAG Key Concepts

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/docs/modules/ROOT/pages/rag.adoc

Defines foundational concepts for Retrieval-Augmented Generation (RAG) applications, including text splitting, embeddings, vector stores, and the retriever component.

```APIDOC
Concept |Description

|Text Splitter |Divides documents into smaller overlapping or non-overlapping segments for processing.
|Embedding |A dense vector representation of a text segment computed by an embedding model.
|Embedding Model |A model trained to convert text into embeddings preserving semantic meaning.
|Vector Store |A database that allows similarity search over embeddings (e.g., PGVector, Chroma, Weaviate).
|Query Embedding |The embedding computed from the user’s question.
|Retriever |A component responsible for finding relevant segments given a query embedding.
|Augmentor |Optional logic used to enhance the retrieved content (e.g., filtering, re-ranking, transformation).
|Prompt Injection |Final step of merging query and retrieved content into a prompt for the model.

```

```Java
    public RAGExample(ChatLanguageModel chatModel, EmbeddingModel embeddingModel, EmbeddingStore<TextSegment> embeddingStore) {
        this.chatModel = chatModel;
        this.embeddingModel = embeddingModel;
        this.retriever = EmbeddingSearchRetriever.builder()
                .embeddingStore(embeddingStore)
                .embeddingModel(embeddingModel)
                .maxResults(1)
                .build();

    public String answerQuestion(String question) {
        List<TextSegment> relevantSegments = retriever.find(question);
        TextSegment segment = relevantSegments.get(0);

        String prompt = String.format("Answer the question based on the following context:\n\nContext:\n%s\n\nQuestion: %s", segment.text(), question);

        Response<ChatMessage> response = chatModel.generate(prompt);
        return response.content().text();
    }
```

### Retrieval Augmented Generation (RAG) with Quarkus Langchain4j

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/integration-tests/simple-ollama/src/main/resources/META-INF/resources/index.html

Illustrates how to implement Retrieval Augmented Generation (RAG) using Langchain4j within Quarkus. This involves setting up document loading, embedding, vector storage, and retrieval mechanisms to enhance LLM responses with external knowledge.

```Java
import dev.langchain4j.data.document.DocumentLoader;
import dev.langchain4j.data.document.FileSystemDocumentLoader;
import dev.langchain4j.data.embedding.Embedding;
import dev.langchain4j.data.segment.TextSegment;
import dev.langchain4j.model.embedding.EmbeddingModel;
import dev.langchain4j.retriever.Retriever;
import dev.langchain4j.store.embedding.EmbeddingStore;
import dev.langchain4j.store.embedding.inmemory.InMemoryEmbeddingStore;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.input.PromptTemplate;
import dev.langchain4j.service.AiServices;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.nio.file.Paths;

@ApplicationScoped
public class RagService {

    @Inject
    ChatLanguageModel chatLanguageModel;
    @Inject
    EmbeddingModel embeddingModel;

    private final EmbeddingStore<TextSegment> embeddingStore = new InMemoryEmbeddingStore<>();
    private final Retriever<TextSegment> retriever;

    public RagService() {
        // Load documents (example: from a file)
        DocumentLoader.load(Paths.get("src/main/resources/documents"), FileSystemDocumentLoader.extensionMatcher("txt"))
                .forEach(document -> {
                    embeddingStore.addAll(document.toTextSegments().stream()
                            .map(segment -> new dev.langchain4j.data.embedding.Embedding(segment, embeddingModel.embed(segment).content()))
                            .toList());
                });
        this.retriever = embeddingStore.asRetriever(10);
    }

    public String askQuestion(String question) {
        String promptTemplate = "Use the following information to answer the question: {{information}}\n\nQuestion: {{question}}\n\nAnswer:";
        String information = retriever.retrieve(question).stream()
                .map(TextSegment::text)
                .reduce("", (a, b) -> a + "\n" + b);

        return chatLanguageModel.generate(PromptTemplate.from(promptTemplate).apply(information, question));
    }
}
```

### Langchain4j RAG Chain Example

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/integration-tests/simple-ollama/src/main/resources/META-INF/resources/index.html

This snippet provides an example of a Retrieval Augmented Generation (RAG) chain using Langchain4j in Quarkus. It demonstrates how to combine a retriever with a language model to answer questions based on provided documents.

```Java
import dev.langchain4j.chain.RAGChain;
import dev.langchain4j.data.document.Document;
import dev.langchain4j.data.document.DocumentLoader;
import dev.langchain4j.data.document.splitter.DocumentSplitter;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.embedding.EmbeddingModel;
import dev.langchain4j.retriever.Retriever;
import dev.langchain4j.store.embedding.EmbeddingStore;
import jakarta.inject.Inject;

import java.util.List;

public class RagChainService {

    @Inject
    private ChatLanguageModel chatLanguageModel;

    @Inject
    private EmbeddingModel embeddingModel;

    @Inject
    private EmbeddingStore<TextSegment> embeddingStore;

    public String askQuestion(String question) {
        // Load and split documents (example)
        Document document = DocumentLoader.fromUrl("http://example.com/document.txt");
        DocumentSplitter splitter = DocumentSplitter.builder().build();
        List<TextSegment> textSegments = splitter.split(document);

        // Embed and store segments
        textSegments.forEach(segment -> {
            Embedding embedding = embeddingModel.embed(segment.text());
            embeddingStore.add(segment, embedding);
        });

        // Create retriever
        Retriever<TextSegment> retriever = EmbeddingStoreRetriever.from(embeddingStore);

        // Create RAG chain
        RAGChain ragChain = RAGChain.builder()
                .chatLanguageModel(chatLanguageModel)
                .retriever(retriever)
                .build();

        return ragChain.execute(question);
    }
}
```

Source: https://github.com/quarkiverse/quarkus-langchain4j/blob/main/docs/modules/ROOT/pages/guardrails.adoc

```java
@ApplicationScoped

    @Inject

    @Override
    public OutputGuardrailResult validate(OutputGuardrailParams params) {
        Response<Embedding> embeddingOfTheResponse = embedding.embed(params.responseFromLLM().text());
        if (params.augmentationResult() == null || params.augmentationResult().contents().isEmpty()) {

            return success();
```
